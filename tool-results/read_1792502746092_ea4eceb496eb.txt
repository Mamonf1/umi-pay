   161→/* ==== تسعير الصرف ==== */
   162→function PricingTab({ me, data, reload }: { me: Me; data: AdminData | null; reload: () => void }) {
   163→  const { rates } = useRates();
   164→  const [usd, setUsd] = useState("");
   165→  const [point, setPoint] = useState("");
   166→  const [msg, setMsg] = useState("");
   167→  const [err, setErr] = useState("");
   168→  const [busy, setBusy] = useState(false);
   169→  const [syncing, setSyncing] = useState(false);
   170→  const [syncData, setSyncData] = useState<{ fetched: number; current: number; source: string; sourceTime: string | null } | null>(null);
   171→
   172→  useEffect(() => {
   173→    if (data) { setUsd(String(Math.round(data.settings.usdSyp))); setPoint(String(Math.round(data.settings.pointSyp))); }
   174→  }, [data]);
   175→
   176→  const save = async () => {
   177→    setBusy(true); setErr(""); setMsg("");
   178→    try {
   179→      await api("/api/umi/admin", {
   180→        method: "PUT",
   181→        body: JSON.stringify({ userId: me.id, usdSyp: Number(usd), pointSyp: Number(point) }),
   182→      });
   183→      setMsg("تم حفظ التسعير ✓");
   184→      reload();
   185→    } catch (e) {
   186→      setErr((e as Error).message);
   187→    } finally {
   188→      setBusy(false);
   189→    }
   190→  };
   191→
   192→  const fetchFromSource = async () => {
   193→    setSyncing(true); setErr(""); setMsg(""); setSyncData(null);
   194→    try {
   195→      const r = await api<{ fetched: number; current: number; source: string; sourceTime: string | null }>("/api/umi/admin/sync-rate", {
   196→        method: "POST",
   197→        body: JSON.stringify({ adminId: me.id }),
   198→      });
   199→      setSyncData(r);
   200→    } catch (e) {
   201→      setErr((e as Error).message);
   202→    } finally {
   203→      setSyncing(false);
   204→    }
   205→  };
   206→
   207→  const applyFetched = async (rate: number) => {
   208→    setSyncing(true); setErr(""); setMsg("");
   209→    try {
   210→      await api("/api/umi/admin/sync-rate", {
   211→        method: "POST",
   212→        body: JSON.stringify({ adminId: me.id, apply: true, rate }),
   213→      });
   214→      setUsd(String(rate));
   215→      setMsg(`تم تطبيق السعر المجلوب: $1 = ${fmtSyp(rate)} ${SYP_UNIT} ✓`);
   216→      setSyncData(null);
   217→      reload();
   218→    } catch (e) {
   219→      setErr((e as Error).message);
   220→    } finally {
   221→      setSyncing(false);
   222→    }
   223→  };
   224→
   225→  const usdPresets = [11000, 12500, 13000, 15000];
   226→  const pointPresets = [5, 10, 25, 50];
   227→
   228→  return (
   229→    <div className="px-4 space-y-3">
   230→      {/* سعر الدولار */}
   231→      <div className="rounded-2xl umi-card p-4">
   232→        <p className="flex items-center gap-1.5 text-[12px] font-cairo font-bold text-umi-cream mb-1">
   233→          <Globe className="w-4 h-4 text-umi-gold" /> سعر الدولار مقابل الليرة
   234→        </p>
   235→        <p className="text-[10px] font-tajawal text-umi-muted mb-3 leading-relaxed">
   236→          يُستخدم لحساب مكافئ الدولار "≈ $" تحت كل رصيد — يمكنك الإدخال اليدوي أو المزامنة من مصدر موثوق
   237→        </p>
   238→
   239→        <button
   240→          type="button"
   241→          onClick={fetchFromSource}
   242→          disabled={syncing}
   243→          className="w-full h-11 rounded-xl font-cairo font-extrabold text-[12.5px] text-umi-cream flex items-center justify-center gap-2 active:scale-[0.98] transition-all disabled:opacity-50"
   244→          style={{ background: "linear-gradient(135deg,#3a2a5e,#2a1c48)", border: "1px solid rgba(232,199,102,0.35)" }}
   245→        >
   246→          <RefreshCw className={`w-4 h-4 text-umi-gold ${syncing ? "animate-spin" : ""}`} />
   247→          {syncing ? "جارٍ الجلب من المصدر الموثوق…" : "جلب سعر الصرف من مصدر موثوق"}
   248→        </button>
   249→
   250→        {/* مقارنة السعر المجلوب بالحالي — التطبيق بقرار الأدمن */}
   251→        {syncData && (
   252→          <div className="mt-3 rounded-xl p-3" style={{ border: "1px solid rgba(111,181,163,0.35)", background: "rgba(111,181,163,0.07)" }}>
   253→            <p className="text-[10.5px] font-tajawal text-umi-teal font-bold mb-1">
   254→              المصدر: {syncData.source} {syncData.sourceTime ? `· آخر تحديث: ${syncData.sourceTime.slice(5, 16)}` : ""}
   255→            </p>
   256→            <div className="flex items-center justify-between text-[11px] font-tajawal text-umi-cream" dir="ltr">
   257→              <span>المجلوب: <b className="text-umi-gold">${syncData.fetched.toLocaleString("en-US")}</b></span>
   258→              <span>الحالي: {syncData.current.toLocaleString("en-US")}</span>
   259→            </div>
   260→            {syncData.fetched !== Math.round(syncData.current) && (
   261→              <p className="text-[9.5px] font-tajawal text-umi-muted mt-1.5 leading-relaxed">
   262→                تنبيه: المصدر يعطي السعر الرسمي وقد يختلف كثيراً عن سعر السوق لديك — طبّقه فقط إذا كان مناسباً
   263→              </p>
   264→            )}
   265→            <div className="flex gap-2 mt-2.5">
   266→              <button
   267→                type="button"
   268→                onClick={() => applyFetched(syncData.fetched)}
   269→                className="flex-1 h-9 rounded-lg font-tajawal font-bold text-[11px] text-[#0a0416]"
   270→                style={{ background: "linear-gradient(135deg,#f5dfa0,#e8c766)" }}
   271→              >
   272→                تطبيق السعر المجلوب
   273→              </button>
   274→              <button
   275→                type="button"
   276→                onClick={() => setSyncData(null)}
   277→                className="flex-1 h-9 rounded-lg font-tajawal font-bold text-[11px] text-umi-muted umi-card"
   278→              >
   279→                إبقاء السعر الحالي
   280→              </button>
   281→            </div>
   282→          </div>
   283→        )}
   284→
   285→        <div className="umi-shimmer-line my-3.5" />
   286→
   287→        <div className="flex gap-2 mb-2">
   288→          {usdPresets.map((p) => (
   289→            <button
   290→              key={p}
   291→              type="button"
   292→              onClick={() => setUsd(String(p))}
   293→              className="flex-1 h-8 rounded-lg umi-card text-[10px] font-tajawal font-bold text-umi-cream active:bg-white/10"
   294→              dir="ltr"
   295→            >
   296→              {p.toLocaleString("en-US")}
   297→            </button>
   298→          ))}
   299→        </div>
   300→        <label className="text-[10.5px] font-tajawal font-bold text-umi-muted block mb-1">إدخال يدوي — $1 = ؟ ل.س</label>
   301→        <input
   302→          value={usd}
   303→          onChange={(e) => setUsd(e.target.value.replace(/\D/g, "").slice(0, 8))}
   304→          inputMode="numeric"
   305→          dir="ltr"
   306→          className="w-full h-10 rounded-xl bg-white/5 border border-white/10 px-3 text-center font-cairo font-bold text-umi-cream outline-none focus:border-umi-gold/40"
   307→        />
   308→        <p className="text-[9.5px] font-tajawal text-umi-muted mt-2 text-center" dir="ltr">
   309→          100,000 {SYP_UNIT} ≈ {fmtUsd(sypToUsd(100000, Number(usd) || rates.usdSyp))}
   310→        </p>
   311→      </div>
   312→
   313→      {/* قيمة نقطة أومي */}
   314→      <div className="rounded-2xl umi-card p-4">
   315→        <p className="flex items-center gap-1.5 text-[12px] font-cairo font-bold text-umi-cream mb-1">
   316→          <Coins className="w-4 h-4 text-umi-gold" /> قيمة نقطة أومي
   317→        </p>
   318→        <p className="text-[10px] font-tajawal text-umi-muted mb-3 leading-relaxed">
   319→          النقاط محفظة مكافآت تُمنح من الإدارة فقط — لا بيع ولا شراء. القيمة هنا للعرض وحساب المكافئ بالليرة
   320→        </p>
   321→        <div className="flex gap-2 mb-2">
   322→          {pointPresets.map((p) => (
   323→            <button
   324→              key={p}
   325→              type="button"
   326→              onClick={() => setPoint(String(p))}
   327→              className="flex-1 h-8 rounded-lg umi-card text-[10px] font-tajawal font-bold text-umi-cream active:bg-white/10"
   328→              dir="ltr"
   329→            >
   330→              {p}
   331→            </button>
   332→          ))}
   333→        </div>
   334→        <label className="text-[10.5px] font-tajawal font-bold text-umi-muted block mb-1">قيمة النقطة الواحدة — ل.س</label>
   335→        <input
   336→          value={point}
   337→          onChange={(e) => setPoint(e.target.value.replace(/\D/g, "").slice(0, 7))}
   338→          inputMode="numeric"
   339→          dir="ltr"
   340→          className="w-full h-10 rounded-xl bg-white/5 border border-white/10 px-3 text-center font-cairo font-bold text-umi-cream outline-none focus:border-umi-gold/40"
   341→        />
   342→        <p className="text-[9.5px] font-tajawal text-umi-muted mt-2 text-center">
   343→          100 نقطة ≈ {fmtSyp(pointsToSyp(100, Number(point) || rates.pointSyp))} {SYP_UNIT}
   344→        </p>
   345→      </div>
   346→
   347→      {err && <p className="text-[11px] font-tajawal text-umi-red text-center">{err}</p>}
   348→      {msg && <p className="text-[11px] font-tajawal text-umi-teal text-center">{msg}</p>}
   349→      <button
   350→        type="button"
   351→        disabled={busy}
   352→        onClick={save}
   353→        className="w-full h-11 rounded-xl font-cairo font-extrabold text-sm text-[#0a0416] disabled:opacity-40 active:scale-[0.98] transition-all flex items-center justify-center gap-2"
   354→        style={{ background: "linear-gradient(135deg,#f5dfa0,#e8c766 55%,#c99b3f)" }}
   355→      >
   356→        <Save className="w-4 h-4" /> {busy ? "جارٍ الحفظ…" : "حفظ التسعير"}
   357→      </button>
   358→    </div>
   359→  );
   360→}
