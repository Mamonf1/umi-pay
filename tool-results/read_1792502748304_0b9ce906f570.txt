   162→   162→function PricingTab({ me, data, reload }: { me: Me; data: AdminData | null; reload: () => void }) {
   163→   163→  const { rates } = useRates();
   164→   164→  const [usd, setUsd] = useState("");
   165→   165→  const [point, setPoint] = useState("");
   166→   166→  const [msg, setMsg] = useState("");
   167→   167→  const [err, setErr] = useState("");
   168→   168→  const [busy, setBusy] = useState(false);
   169→   169→  const [syncing, setSyncing] = useState(false);
   170→   170→  const [syncData, setSyncData] = useState<{ fetched: number; current: number; source: string; sourceTime: string | null } | null>(null);
   171→   171→
   172→   172→  useEffect(() => {
   173→   173→    if (data) { setUsd(String(Math.round(data.settings.usdSyp))); setPoint(String(Math.round(data.settings.pointSyp))); }
   174→   174→  }, [data]);
   175→   175→
   176→   176→  const save = async () => {
   177→   177→    setBusy(true); setErr(""); setMsg("");
   178→   178→    try {
   179→   179→      await api("/api/umi/admin", {
   180→   180→        method: "PUT",
   181→   181→        body: JSON.stringify({ userId: me.id, usdSyp: Number(usd), pointSyp: Number(point) }),
   182→   182→      });
   183→   183→      setMsg("تم حفظ التسعير ✓");
   184→   184→      reload();
   185→   185→    } catch (e) {
   186→   186→      setErr((e as Error).message);
   187→   187→    } finally {
   188→   188→      setBusy(false);
   189→   189→    }
   190→   190→  };
   191→   191→
   192→   192→  const fetchFromSource = async () => {
   193→   193→    setSyncing(true); setErr(""); setMsg(""); setSyncData(null);
   194→   194→    try {
   195→   195→      const r = await api<{ fetched: number; current: number; source: string; sourceTime: string | null }>("/api/umi/admin/sync-rate", {
   196→   196→        method: "POST",
   197→   197→        body: JSON.stringify({ adminId: me.id }),
   198→   198→      });
   199→   199→      setSyncData(r);
   200→   200→    } catch (e) {
   201→   201→      setErr((e as Error).message);
   202→   202→    } finally {
   203→   203→      setSyncing(false);
   204→   204→    }
   205→   205→  };
   206→   206→
   207→   207→  const applyFetched = async (rate: number) => {
   208→   208→    setSyncing(true); setErr(""); setMsg("");
   209→   209→    try {
   210→   210→      await api("/api/umi/admin/sync-rate", {
   211→   211→        method: "POST",
   212→   212→        body: JSON.stringify({ adminId: me.id, apply: true, rate }),
   213→   213→      });
   214→   214→      setUsd(String(rate));
   215→   215→      setMsg(`تم تطبيق السعر المجلوب: $1 = ${fmtSyp(rate)} ${SYP_UNIT} ✓`);
   216→   216→      setSyncData(null);
   217→   217→      reload();
   218→   218→    } catch (e) {
   219→   219→      setErr((e as Error).message);
   220→   220→    } finally {
   221→   221→      setSyncing(false);
   222→   222→    }
   223→   223→  };
   224→   224→
   225→   225→  const usdPresets = [11000, 12500, 13000, 15000];
   226→   226→  const pointPresets = [5, 10, 25, 50];
   227→   227→
   228→   228→  return (
   229→   229→    <div className="px-4 space-y-3">
   230→   230→      {/* سعر الدولار */}
   231→   231→      <div className="rounded-2xl umi-card p-4">
   232→   232→        <p className="flex items-center gap-1.5 text-[12px] font-cairo font-bold text-umi-cream mb-1">
   233→   233→          <Globe className="w-4 h-4 text-umi-gold" /> سعر الدولار مقابل الليرة
   234→   234→        </p>
   235→   235→        <p className="text-[10px] font-tajawal text-umi-muted mb-3 leading-relaxed">
   236→   236→          يُستخدم لحساب مكافئ الدولار "≈ $" تحت كل رصيد — يمكنك الإدخال اليدوي أو المزامنة من مصدر موثوق
   237→   237→        </p>
   238→   238→
   239→   239→        <button
   240→   240→          type="button"
   241→   241→          onClick={fetchFromSource}
   242→   242→          disabled={syncing}
   243→   243→          className="w-full h-11 rounded-xl font-cairo font-extrabold text-[12.5px] text-umi-cream flex items-center justify-center gap-2 active:scale-[0.98] transition-all disabled:opacity-50"
   244→   244→          style={{ background: "linear-gradient(135deg,#3a2a5e,#2a1c48)", border: "1px solid rgba(232,199,102,0.35)" }}
   245→   245→        >
   246→   246→          <RefreshCw className={`w-4 h-4 text-umi-gold ${syncing ? "animate-spin" : ""}`} />
   247→   247→          {syncing ? "جارٍ الجلب من المصدر الموثوق…" : "جلب سعر الصرف من مصدر موثوق"}
   248→   248→        </button>
   249→   249→
   250→   250→        {/* مقارنة السعر المجلوب بالحالي — التطبيق بقرار الأدمن */}
   251→   251→        {syncData && (
   252→   252→          <div className="mt-3 rounded-xl p-3" style={{ border: "1px solid rgba(111,181,163,0.35)", background: "rgba(111,181,163,0.07)" }}>
   253→   253→            <p className="text-[10.5px] font-tajawal text-umi-teal font-bold mb-1">
   254→   254→              المصدر: {syncData.source} {syncData.sourceTime ? `· آخر تحديث: ${syncData.sourceTime.slice(5, 16)}` : ""}
   255→   255→            </p>

   259→   259→            </div>
